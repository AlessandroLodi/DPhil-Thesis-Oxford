\section{Additional Data}\label{sec:appendix:gnr_hybrid}
Here below we report additional data for the \glspl{pGNR}

\begin{figure}[hbtp]
    \begin{center}
        \includegraphics[width=\textwidth]{text/b11_c4.pdf}
        \caption[MGNR--Ni Hybrid: Additional FETs (device b11, c4)]
        {
            \textbf{left,} \gls{FET} characteristics:  $I_\textrm{on}/I_\textrm{off}\simeq 10^2$ at $V_\textrm{SD}=\SI{0.1}{\volt}$, ambipolar behaviour at  $V_\textrm{SD}=\SI{0.5}{\volt}$, $SS\simeq\SI{430}{\milli\volt}$~dec$^{-1}$ at $V_\textrm{SD}=\SI{0.5}{\volt}$.

            \textbf{right,} \gls{FET} characteristics:  $I_\textrm{on}/I_\textrm{off}\simeq 10^2$ at $V_\textrm{SD}=\SI{0.3}{\volt}$, small hysteresis present in the $V_\textrm{SD}=\SI{0.3}{\volt}$ transfer characteristics, $SS\simeq\SI{435}{\milli\volt}$~dec$^{-1}$ at $V_\textrm{SD}=\SI{0.3}{\volt}$ for the forward sweep, degrades down to $SS\simeq\SI{870}{\milli\volt}$~dec$^{-1}$ for the backward sweep, $\mu_\textrm{lin}=\SI{11}{\centi\metre\squared\per\volt\per\second}$, $\mu_\textrm{sat}=\SI{4.3}{\centi\metre\squared\per\volt\per\second}$
        }
        \label{fig:spinvalve_structure_geom_optimised}
    \end{center}
\end{figure}
